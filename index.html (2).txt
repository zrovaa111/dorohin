<script>
var activeCategory = 'all';
var maxPrice = 5000;

function filterByPrice(val) {
  maxPrice = parseInt(val);
  var display = maxPrice + ' ₽';
  document.getElementById('priceDisplay').textContent = display;
  var pct = (maxPrice / 5000) * 100;
  document.getElementById('sliderFill').style.width = pct + '%';
  applyFilters();
}

function filterPortfolio(cat, btn) {
  var btns = document.querySelectorAll('.subtab');
  for (var i = 0; i < btns.length; i++) btns[i].classList.remove('active');
  btn.classList.add('active');
  activeCategory = cat;
  applyFilters();
}

function applyFilters() {
  var cards = document.querySelectorAll('#portfolioGrid .pf-card');
  for (var i = 0; i < cards.length; i++) {
    var card = cards[i];
    var catMatch = activeCategory === 'all' || card.getAttribute('data-cat') === activeCategory;
    var price = parseInt(card.getAttribute('data-price') || '0');
    var priceMatch = price <= maxPrice;
    card.style.display = (catMatch && priceMatch) ? '' : 'none';
  }
}

function showTab(name, btn) {
  var sections = document.querySelectorAll('.section');
  for (var i = 0; i < sections.length; i++) sections[i].classList.remove('active');
  var tabs = document.querySelectorAll('.tab-btn');
  for (var j = 0; j < tabs.length; j++) tabs[j].classList.remove('active');
  document.getElementById('tab-' + name).classList.add('active');
  btn.classList.add('active');
  var nav = document.querySelector('.tab-nav');
  if (nav) window.scrollTo(0, nav.offsetTop - 1);
}

function toggleFaq(item) {
  var isOpen = item.classList.contains('open');
  var all = document.querySelectorAll('.faq-item');
  for (var i = 0; i < all.length; i++) all[i].classList.remove('open');
  if (!isOpen) item.classList.add('open');
}

function acceptTerms() {
  var btn = document.getElementById('acceptBtn');
  var bar = document.getElementById('acceptBar');
  var block = document.getElementById('accept');
  var textEl = document.getElementById('acceptBtnText');

  bar.style.width = '100%';
  btn.classList.add('done-btn');
  block.classList.add('done');
  textEl.innerHTML = 'Отлично! Открываю Telegram...';

  var msg = encodeURIComponent('Привет! Ознакомился с правилами, готов работать');
  window.open('https://t.me/zroven?text=' + msg, '_blank');
}

// init slider
